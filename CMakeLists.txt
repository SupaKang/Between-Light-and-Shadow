cmake_minimum_required(VERSION 3.20)
project(YinYangChronicle LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BLS_SANITIZE "Build with AddressSanitizer + UBSan" OFF)
if(NOT MSVC)
  add_compile_options(-Wall -Wextra)
  if(BLS_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_executable(YinYangChronicle WIN32 src/main.cpp src/battle_rules.cpp src/status_rules.cpp src/battle_state.cpp src/collection_state.cpp src/data_registry.cpp src/battle_ui.cpp src/save_rules.cpp src/skill_state.cpp src/world_state.cpp)
add_executable(BattleRulesTest tests/battle_rules_test.cpp src/battle_rules.cpp src/status_rules.cpp src/battle_state.cpp src/collection_state.cpp src/data_registry.cpp src/battle_ui.cpp src/save_rules.cpp src/skill_state.cpp src/world_state.cpp)
add_test(NAME BattleRulesTest COMMAND BattleRulesTest)
target_compile_definitions(YinYangChronicle PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(YinYangChronicle PRIVATE /W4 /O2 /GL /utf-8)
  target_link_options(YinYangChronicle PRIVATE /LTCG /OPT:REF /OPT:ICF)
else()
  find_package(PkgConfig QUIET)
  if(PKG_CONFIG_FOUND)
    pkg_check_modules(SDL2 QUIET sdl2)
    if(SDL2_FOUND)
      target_compile_definitions(YinYangChronicle PRIVATE USE_SDL2)
      target_include_directories(YinYangChronicle PRIVATE ${SDL2_INCLUDE_DIRS})
      target_link_directories(YinYangChronicle PRIVATE ${SDL2_LIBRARY_DIRS})
      target_link_libraries(YinYangChronicle PRIVATE ${SDL2_LIBRARIES})
    endif()
  endif()
endif()
